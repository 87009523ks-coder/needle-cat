    <script type="module">

        import { initializeApp } from "https://www.gstatic.com/firebasejs/10.7.1/firebase-app.js";

        import { getFirestore, collection, addDoc, onSnapshot, query, orderBy } from "https://www.gstatic.com/firebasejs/10.7.1/firebase-firestore.js";


        // 이미지 속 설정값 적용

        const firebaseConfig = {

            apiKey: "AIzaSyBdnFr8vjNOfOeovvFm9UmxvkEpvnuAWLY",

            authDomain: "needle-cat.firebaseapp.com",

            projectId: "needle-cat",

            storageBucket: "needle-cat.appspot.com",

            messagingSenderId: "422542027971",

            appId: "1:422542027971:web:37f2298a24b2161209c676"

        };


        const app = initializeApp(firebaseConfig);

        const db = getFirestore(app);

        let myChart;


        // 저장 함수

        window.saveData = async () => {

            const date = document.getElementById('inputDate').value;

            const delivery = document.getElementById('inputDelivery').value;

            const material = document.getElementById('inputMaterial').value;

            const price = document.getElementById('inputPrice').value;


            if(!date || !delivery || !price) return alert("빈칸을 채워주세요!");


            await addDoc(collection(db, "sales"), {

                date: date,

                delivery: parseInt(delivery),

                material: parseInt(material || 0),

                price: parseInt(price),

                timestamp: new Date()

            });

            alert("저장 성공!");

            document.getElementById('inputDelivery').value = '';

            document.getElementById('inputMaterial').value = '';

            document.getElementById('inputPrice').value = '';

        };


        // 데이터 실시간 감시 및 업데이트

        const q = query(collection(db, "sales"), orderBy("date", "asc"));

        onSnapshot(q, (snapshot) => {

            let totalD = 0; let totalP = 0;

            let labels = []; let counts = [];

            const historyList = document.getElementById('historyList');

            historyList.innerHTML = '';


            const docs = snapshot.docs.map(doc => doc.data());

            

            docs.forEach(data => {

                totalD += data.delivery;

                totalP += data.price;

                labels.push(data.date.substring(5)); // MM-DD만 차트에 표시

                counts.push(data.delivery);


                // 리스트 추가 (최신순으로 보이게 앞쪽에 추가)

                const item = `<div class="history-item">

                    <div>

                        <div class="history-date">${data.date}</div>

                        <div style="font-size:15px;">납품 ${data.delivery}개 / 재료 ${data.material}개</div>

                    </div>

                    <div style="font-weight:700;">₩${data.price.toLocaleString()}</div>

                </div>`;

                historyList.insertAdjacentHTML('afterbegin', item);

            });


            document.getElementById('totalDelivery').innerText = totalD.toLocaleString();

            document.getElementById('totalPrice').innerText = totalP.toLocaleString();

            updateChart(labels, counts);

        });


        function updateChart(labels, data) {

            const ctx = document.getElementById('salesChart').getContext('2d');

            if(myChart) myChart.destroy();

            myChart = new Chart(ctx, {

                type: 'line',

                data: {

                    labels: labels,

                    datasets: [{

                        label: '판매량',

                        data: data,

                        borderColor: '#2D3436',

                        tension: 0.4,

                        fill: false

                    }]

                },

                options: { plugins: { legend: { display: false } }, scales: { y: { beginAtZero: true } } }

            });

        }
